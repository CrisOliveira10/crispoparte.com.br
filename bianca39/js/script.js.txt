/* =========================
   AUDIO PLAY / PAUSE
========================= */

const audio = document.getElementById("bgAudio");
const audioBtn = document.getElementById("audioBtn");
const audioIcon = document.getElementById("audioIcon");

let tocando = false;

audioBtn.addEventListener("click", () => {

  if (!tocando) {

    audio.play().then(() => {
      audioIcon.textContent = "❚❚";
      tocando = true;
    }).catch(() => {
      console.log("Autoplay bloqueado pelo navegador");
    });

  } else {

    audio.pause();
    audioIcon.textContent = "▶";
    tocando = false;

  }

});


/* =========================
   CONTADOR
========================= */

const dataEvento = new Date("2026-02-21T20:00:00");

function atualizarContador() {

  const agora = new Date();
  const diff = dataEvento - agora;

  const countdown = document.getElementById("countdown");

  if (!countdown) return;

  if (diff <= 0) {
    countdown.innerHTML = "É HOJE!";
    return;
  }

  const dias = Math.floor(diff / (1000 * 60 * 60 * 24));
  const horas = Math.floor((diff / (1000 * 60 * 60)) % 24);
  const minutos = Math.floor((diff / (1000 * 60)) % 60);
  const segundos = Math.floor((diff / 1000) % 60);

  countdown.innerHTML =
    `${dias}d ${horas}h ${minutos}m ${segundos}s`;
}

atualizarContador();
setInterval(atualizarContador, 1000);


/* =========================
   GERAR ARQUIVO ICS
========================= */

function gerarICS() {

  const inicio = "20260221T200000";
  const fim = "20260221T230000";

  const ics =
`BEGIN:VCALENDAR
VERSION:2.0
BEGIN:VEVENT
DTSTART;TZID=America/Sao_Paulo:${inicio}
DTEND;TZID=America/Sao_Paulo:${fim}
SUMMARY:Aniversário Bianca 39
DESCRIPTION:Venha celebrar comigo!
LOCATION:MilMar Gelateria e Cafeteria - Salvador BA
END:VEVENT
END:VCALENDAR`;

  const blob = new Blob([ics], { type: "text/calendar" });
  const link = document.createElement("a");
  link.href = URL.createObjectURL(blob);
  link.download = "convite.ics";
  link.click();
}


/* =========================
   MODAL
========================= */

function abrirModal() {
  document.getElementById("modal").style.display = "flex";
}

function fecharModal() {
  document.getElementById("modal").style.display = "none";
}

window.addEventListener("click", function(e){
  const modal = document.getElementById("modal");
  if(e.target === modal){
    modal.style.display = "none";
  }
});


/* =========================
   COPIAR PIX (ATUALIZADO)
========================= */

const copyButton = document.getElementById("copyButton");

if (copyButton) {

  copyButton.addEventListener("click", function () {

    const pixCode = document.getElementById("pixCode").innerText;
    const message = document.getElementById("copyMessage");

    navigator.clipboard.writeText(pixCode)
      .then(() => {

        message.style.display = "block";

        setTimeout(() => {
          message.style.display = "none";
        }, 2000);

      })
      .catch(() => {
        alert("Não foi possível copiar. Copie manualmente.");
      });

  });

}
